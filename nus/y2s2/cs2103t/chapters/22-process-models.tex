\section{Process models}

% ── When to pick each ────────────────────────────────────────────────────────
\subsection{When to pick which?}
\textit{(Names on ref sheet; the decision criteria are not.)}
\begin{itemize}
  \item \textcolor{Bittersweet}{Sequential (waterfall)}: requirements stable + well-understood; effort estimable; progress tracked by stage.
  \item \textcolor{Bittersweet}{Iterative — breadth-first}: working product needed each iteration; risk: shallow first cuts.
  \item \textcolor{Bittersweet}{Iterative — depth-first}: high-risk components needed first; risk: no working product early.
  \item \textcolor{Bittersweet}{Agile (XP, Scrum)}: requirements volatile; close customer collaboration; tolerate evolving design.
  \item \textcolor{Bittersweet}{Unified Process}: large project needing customisable iteration count + phase intensities.
\end{itemize}
\textcolor{ForestGreen}{Mix is normal}: depth-first early iterations $\to$ breadth-first later; sequential within iterations.

% ── Scrum mechanics ───────────────────────────────────────────────────────────
\subsection{Scrum mechanics (MCQ trap)}
\begin{itemize}
  \item \textcolor{Bittersweet}{Roles}: Scrum Master (process), Product Owner (stakeholders), Team (cross-functional).
  \item \textcolor{Bittersweet}{Sprint}: fixed length (1 wk–1 mo). Backlog \textbf{freezes} once sprint starts.
  \item \textcolor{Bittersweet}{Timeboxed}: unfinished features return to backlog, never spill into next sprint.
  \item \textcolor{Bittersweet}{Daily scrum}: 15 min, 3 questions: did yesterday / will do today / impediments. Issues resolved \emph{offline} after, not in the meeting.
\end{itemize}

% ── UP phases ────────────────────────────────────────────────────────────────
\paragraph{Unified Process: 4 phases}
\textcolor{Bittersweet}{Inception} (vision, rough plan) $\to$ \textcolor{Bittersweet}{Elaboration} (architecture, design) $\to$ \textcolor{Bittersweet}{Construction} (implementation, testing) $\to$ \textcolor{Bittersweet}{Transition} (deploy, train users).

% ── Agile manifesto ──────────────────────────────────────────────────────────
\paragraph{Agile manifesto: 4 values}
Individuals + interactions $>$ processes/tools; working software $>$ docs; customer collab $>$ contract; respond to change $>$ follow plan.
