\section{Renewal Theory}
\subsection{Setup}
A \textcolor{Blue}{\textbf{renewal process}} has iid non-negative interarrivals
$X_i \sim F$ with mean $\mu = \mathbb{E}[X_1]$ and variance $\sigma^2$. Define
\textcolor{Blue}{\textbf{epochs}} $S_0 = 0$, $S_n = \sum_{i=1}^{n} X_i$, and
the \textcolor{Blue}{\textbf{counting process}}
\[N(t) = \sup\{n \ge 0 : S_n \le t\}.\]
\textcolor{Bittersweet}{Fundamental relation}: $\{N(t) \ge k\} \iff \{S_k \le t\}$, so
\[\mathbb{P}(N(t) = k) = F_k(t) - F_{k+1}(t),\]
where $F_k$ is the $k$-fold convolution of $F$. The
\textcolor{Blue}{\textbf{renewal function}} is
\[m(t) = \mathbb{E}[N(t)] = \sum_{k \ge 1} F_k(t).\]
\textcolor{Bittersweet}{SLLN for renewals}: $N(t)/t \to 1/\mu$ a.s. as $t \to
\infty$, for $\mu < \infty$.\\
\textcolor{ForestGreen}{Poisson special case}: $X_i \sim \mathrm{Exp}(\lambda)$
gives $S_k \sim \mathrm{Gamma}(k, \lambda)$ and $m(t) = \lambda t$.

\subsection{Renewal equation \& main theorems}
\paragraph{Renewal equation} Conditioning on $X_1$:
\[m(t) = F(t) + \int_0^t m(t-s)\,dF(s).\]
Equivalent convolution form: $m(t) = F(t) + \int_0^t F(t-y)\,dm(y)$.\\
\textcolor{Bittersweet}{Laplace transform}: $\tilde m(s) = \tilde F(s)/(s(1 -
\tilde F(s)))$.

\paragraph{Asymptotic theorems}
\begin{itemize}
  \item \textcolor{Bittersweet}{Elementary renewal theorem (ERT)}: $m(t)/t \to
    1/\mu$.
  \item \textcolor{Bittersweet}{Asymptotic variance}: $\mathrm{Var}(N(t)) \sim
    \sigma^2 t / \mu^3$ for $\sigma^2 < \infty$.
  \item \textcolor{Bittersweet}{Blackwell's theorem} ($F$ non-arithmetic, $\mu
    < \infty$): for any $h > 0$,
    \[m(t+h) - m(t) \to h/\mu.\]
    Arithmetic span $d$: $m(nd) \sim nd/\mu$.
  \item \textcolor{Bittersweet}{Key renewal theorem (KRT)}: if $g$ is
    \textcolor{Bittersweet}{directly Riemann integrable (DRI)} on $[0,\infty)$
    and $h(t) = g(t) + \int_0^t h(t-s)\,dF(s)$, then $h(t) = \int_0^t
    g(t-s)\,dm(s)$ and
    \[h(t) \to \frac{1}{\mu}\int_0^\infty g(u)\,du.\]
  \item \textcolor{Bittersweet}{CLT}: for $\sigma^2 < \infty$,
    \[\frac{N(t) - t/\mu}{\sqrt{\sigma^2 t/\mu^3}} \xrightarrow{d}
    \mathcal{N}(0,1).\]
\end{itemize}

\subsection{Renewal-reward}
A \textcolor{Blue}{\textbf{renewal-reward}} process has iid pairs $(X_i, R_i)$
(within a pair $X_i, R_i$ may be dependent) with cumulative reward
\[R(t) = \sum_{i=1}^{N(t)} R_i.\]
\textcolor{Bittersweet}{Renewal-reward theorem}: under $\mathbb{E}|R_1|, \mu <
\infty$,
\[\frac{R(t)}{t} \xrightarrow{a.s.} \frac{\mathbb{E}[R_1]}{\mathbb{E}[X_1]},
\qquad \frac{\mathbb{E}[R(t)]}{t} \to \frac{\mathbb{E}[R_1]}{\mathbb{E}[X_1]}.\]
\textcolor{Bittersweet}{Sketch}: $R(t)/t = (R(t)/N(t))(N(t)/t) \to
\mathbb{E}[R_1] \cdot 1/\mu$.

\subsection{Delayed \& stationary renewal}
\textcolor{Blue}{\textbf{Delayed renewal}}: first epoch $Y \sim G$
(arbitrary), $X_2, X_3, \ldots$ iid $F$. Then
\[m_G(t) = G(t) + \int_0^t m_F(t-s)\,dG(s),\qquad m_G(t)/t \to 1/\mu.\]
The asymptotic rate is unaffected by the initial delay.

\paragraph{Equilibrium distribution} The
\textcolor{Bittersweet}{equilibrium distribution} of $F$ is
\[F_e(x) = \frac{1}{\mu}\int_0^x \bar F(y)\,dy,\qquad \text{density } \bar
F(x)/\mu,\]
where $\bar F = 1 - F$.

\textcolor{Blue}{\textbf{Stationary renewal}}: take $Y \sim F_e$. Then
$\{N(t+s) - N(s)\}_{t\ge 0}$ has the same law for every $s \ge 0$, and the
renewal function is \emph{exact} (not asymptotic):
\[m_s(t) = \frac{t}{\mu} + \frac{\mathbb{E}[X^2]}{2\mu^2} - 1.\]

\subsection{Age, excess, spread}
For an inspection at time $t$:
\begin{itemize}
  \item \textcolor{Blue}{\textbf{Age}} $A(t) = t - S_{N(t)}$ — time since last
    renewal.
  \item \textcolor{Blue}{\textbf{Excess}} (residual life) $Y(t) = S_{N(t)+1} -
    t$ — time until next renewal.
  \item \textcolor{Blue}{\textbf{Spread}} $L(t) = X_{N(t)+1} = A(t) + Y(t)$ — length of
    the cycle covering $t$.
\end{itemize}
\paragraph{Limiting distributions ($F$ non-arithmetic)} Both age and excess
converge in distribution to the equilibrium distribution:
\[A(t),\, Y(t) \xrightarrow{d} F_e,\qquad \text{density } \bar F(x)/\mu.\]

\paragraph{Inspection paradox}
\[\mathbb{E}[L(t)] \to \frac{\mathbb{E}[X^2]}{\mu} = \mu + \frac{\sigma^2}{\mu}
\;\ge\; \mu,\]
with equality iff $X_1$ is degenerate. Intuitively, a random inspection time is
\textcolor{Bittersweet}{length-biased} — more likely to fall inside a long
cycle than a short one.

\subsection{Alternating renewal \& regenerative}
\textcolor{Blue}{\textbf{Alternating renewal}}: alternating ON/OFF cycles with
means $\mu_1,\mu_2$. Long-run ON-fraction
\[\lim_{t\to\infty}\mathbb{P}(\text{ON at }t)=\frac{\mu_1}{\mu_1+\mu_2}.\]
\textcolor{Bittersweet}{Regenerative process}: $\{X(t)\}$ regenerates at epoch
$X_1$ (cycle length); for measurable $f$,
\[\frac{1}{t}\!\int_0^t\! f(X(s))\,ds\xrightarrow{a.s.}
\frac{\mathbb{E}\!\left[\int_0^{X_1}\! f(X(s))\,ds\right]}{\mu}.\]
\textcolor{Bittersweet}{Semi-Markov (gloss)}: holding-time distribution $F_{ij}$
depends on (current $i$, next $j$) state; embedded jump chain governs transitions.
