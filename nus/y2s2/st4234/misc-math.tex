\section{Quick reference}

\paragraph{Conditional moments}
\begin{itemize}
  \item \textcolor{Bittersweet}{Tower}: $\mathbb{E}[X]=\mathbb{E}[\mathbb{E}[X\mid Y]]$.
  \item \textcolor{Bittersweet}{Total variance}: $\mathrm{Var}(X)=\mathbb{E}[\mathrm{Var}(X\mid Y)]+\mathrm{Var}(\mathbb{E}[X\mid Y])$.
  \item \textcolor{Bittersweet}{Jensen}: $f$ convex $\Rightarrow f(\mathbb{E}[X])\le \mathbb{E}[f(X)]$.
\end{itemize}

\paragraph{Gamma / Beta identities}
\[\Gamma(\alpha)=\int_0^\infty t^{\alpha-1}e^{-t}\,dt,\quad \Gamma(n)=(n-1)!\]
\[\Gamma(\alpha+1)=\alpha\Gamma(\alpha).\]
\[\int_0^\infty t^{\alpha-1}e^{-\beta t}\,dt=\tfrac{\Gamma(\alpha)}{\beta^{\alpha}}\ (\beta>0).\]
\[B(\alpha,\beta)=\int_0^1 t^{\alpha-1}(1-t)^{\beta-1}\,dt=\tfrac{\Gamma(\alpha)\Gamma(\beta)}{\Gamma(\alpha+\beta)}.\]

\paragraph{Change of variables}
$Y=g(X)$, $g$ smooth invertible: $f_Y(y)=f_X(g^{-1}(y))\,\bigl|\tfrac{dg^{-1}}{dy}\bigr|$.
\textcolor{ForestGreen}{Linear:} $Y=aX\Rightarrow f_Y(y)=\tfrac{1}{|a|}f_X(y/a)$.
\textcolor{ForestGreen}{Reciprocal:} $Y=1/X\Rightarrow f_Y(y)=\tfrac{1}{y^{2}}f_X(1/y)$.

\paragraph{Sum of squares}
\[\sum_{i=1}^{n}(x_i-\mu)^{2}=\sum_{i=1}^{n}(x_i-\bar{x})^{2}+n(\bar{x}-\mu)^{2}.\]
\textcolor{Bittersweet}{Use:} splits a Normal log-likelihood into a $\mu$-free SSR and a quadratic in $\mu$ — feeds directly into the Normal / Gamma–Normal posteriors.

\paragraph{Completing the square}
\[a\mu^{2}-2b\mu+c=a\!\left(\mu-\tfrac{b}{a}\right)^{2}+\!\left(c-\tfrac{b^{2}}{a}\right).\]
\textcolor{Bittersweet}{Use:} a Normal kernel $\exp\!\bigl(-\tfrac{1}{2}(a\mu^{2}-2b\mu)\bigr)$ is $\mathcal{N}(b/a,\,1/a)$ in $\mu$; the leftover $c-b^{2}/a$ is the marginal-likelihood factor.

\textcolor{Bittersweet}{Notes:}
\begin{itemize}
  \item Dir marginals: $X_i\sim\mathrm{Beta}(\alpha_i,\alpha_\cdot-\alpha_i)$ with $\alpha_\cdot=\sum_j\alpha_j$.
  \item Mult cross-moments: $\mathrm{Cov}(X_i,X_j)=-np_ip_j$ ($i\ne j$).
  \item Geom / NegBin convention here: failures before $r$th success ($r=1$ for Geom).
  \item Conj. partner is the \emph{other} family in the conjugate pairing --- read row-by-row whether the family acts as prior or likelihood.
\end{itemize}
